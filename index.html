async function loadBlog() { try { const rss = "https://musicnewsid.blogspot.com/feeds/posts/default?alt=rss"; // ⚡ paksa anti-cache + bypass RSS2JSON cache const api = "https://api.rss2json.com/v1/api.json?rss_url=" + encodeURIComponent(rss) + "&nocache=" + Date.now(); const res = await fetch(api, { cache: "no-store" }); const data = await res.json(); let html = ""; if (!data.items || data.items.length === 0) { document.getElementById("news").innerHTML = "Belum ada artikel."; return; } // 🔥 pastikan urutan terbaru di atas data.items.sort((a, b) => new Date(b.pubDate) - new Date(a.pubDate)); data.items.forEach(post => { let title = post.title || "Tanpa Judul"; let desc = post.description ? post.description.replace(/<[^>]*>/g, '') : ""; let link = post.link; html += `

${title}

${desc.substring(0, 140)}...

Baca Selengkapnya →
`; }); document.getElementById("news").innerHTML = html; } catch (err) { console.log(err); document.getElementById("news").innerHTML = "Gagal memuat artikel."; } } loadBlog();
